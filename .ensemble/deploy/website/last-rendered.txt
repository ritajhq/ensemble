services:
  server:
    image: 'website:latest'
    environment:
      DOCS_DIR: /data/docs
      DOCS_REPOSITORY: ritajhq/ensemble
      DOCS_REF: main
      DOCS_POLL_SECONDS: '300'
      SITE_HOST: ensemble.blue
      DOCS_HOST: docs.ensemble.blue
    volumes:
      - 'docs-content:/data'
    develop:
      watch:
        - path: website/content
          target: /app/content
          action: sync
          initial_sync: true
        - path: website/server
          target: /app/server
          action: sync+restart
          initial_sync: true
    restart: always
  gateway:
    image: 'caddy:2.11-alpine'
    networks:
      edge:
        aliases:
          - '${COMPOSE_PROJECT_NAME}'
      default: null
    labels:
      caddy: 'ensemble.blue, docs.ensemble.blue'
      caddy.tls: internal
      caddy.reverse_proxy: '{{upstreams 80}}'
    configs:
      - source: gateway-caddyfile
        target: /etc/caddy/Caddyfile
    volumes:
      - 'gateway-data:/data'
volumes:
  docs-content: {}
  gateway-data: {}
networks:
  edge:
    external: true
configs:
  gateway-caddyfile:
    content: "http://ensemble.blue {\n\thandle /* {\n\t\treverse_proxy http://server:8000\n\t}\n}\n\nhttp://docs.ensemble.blue {\n\thandle /* {\n\t\treverse_proxy http://server:8000\n\t}\n}\n"
